# Needs the whisper.cpp compose profile (a model download and minutes of CPU).

# search service at all, so this stack cannot satisfy it. Runs in

# because it is finished.
set E2E_SEARCH_SERVICE=true

# QUARANTINE_NEW_UPLOADS is off on this stack, so there is no held upload to
# approve or reject. Runs in frontend-e2e-optional.yml with it on.
